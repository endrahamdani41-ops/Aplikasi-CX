Pengaturan Google Sheets (Penyimpanan)

Rekomendasi

Masukkan URL Google Apps Script Web App untuk menyimpan presensi langsung ke Google Sheets secara otomatis.

Status koneksi Google Sheets akan tampil di sini.
Skrip Google Apps Script

Tempel skrip ini pada menu Ekstensi > Apps Script di dokumen Google Sheets Anda.

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet();
    var data = JSON.parse(e.postData.contents);
    if (data.action === "add_attendance") {
      var attSheet = sheet.getSheetByName("Presensi") || sheet.insertSheet("Presensi");
      if (attSheet.getLastRow() === 0) attSheet.appendRow(["NISN", "Nama Siswa", "Kelas", "Tanggal", "Jam Hadir", "Status"]);
      attSheet.appendRow(["'" + data.nisn, data.name, data.class_name, data.date, data.time, data.status]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
    if (data.action === "enroll_student") {
      var stdSheet = sheet.getSheetByName("Siswa") || sheet.insertSheet("Siswa");
      if (stdSheet.getLastRow() === 0) stdSheet.appendRow(["ID", "NISN", "Nama Siswa", "Kelas", "Face Descriptor", "Tanggal Terdaftar"]);
      var wibDate = Utilities.formatDate(new Date(), "Asia/Jakarta", "dd/MM/yyyy HH:mm:ss");
      stdSheet.appendRow([data.id, "'" + data.nisn, data.name, data.class_name, JSON.stringify(data.face_descriptor), wibDate]);
      return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet();
  var stdSheet = sheet.getSheetByName("Siswa");
  var students = [];
  if (stdSheet) {
    var data = stdSheet.getDataRange().getValues();
    for (var i = 1; i < data.length; i++) {
      if (data[i][0]) {
        students.push({
          id: String(data[i][0]),
          nisn: String(data[i][1]).replace(/^'/, ''),
          name: data[i][2],
          class_name: data[i][3],
          face_descriptor: JSON.parse(data[i][4] || "[]")
        });
      }
    }
  }
  return ContentService.createTextOutput(JSON.stringify({ students: students })).setMimeType(ContentService.MimeType.JSON);
}
